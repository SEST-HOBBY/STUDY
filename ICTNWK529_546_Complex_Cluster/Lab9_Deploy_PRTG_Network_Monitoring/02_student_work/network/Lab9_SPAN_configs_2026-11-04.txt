! ============================================================================
! Lab 9 - Implement Local SPAN (Netacad CN 5.3.2.3) - configs mapped to our rack
! Written 2026-11-04 from the brief. Only what the lab asks for; nothing from Lab 8.
! ============================================================================
! LAB TOPOLOGY -> OUR DEVICES
!   R1  = the 1941 labelled ISP (hostname becomes R1)      G0/1 192.168.1.1/24
!   S1  = the 2960 Plus, marker MG-S1 (hostname S1)        VLAN 1 192.168.1.2/24   <- SPAN happens here
!   S3  = the 2960 labelled BNES1 (hostname S3)            VLAN 1 192.168.1.3/24
!   PC-A = your PC (Wireshark)                              192.168.1.254/24 gw .1
!   PC-C = classmate's PC                                   192.168.1.10/24  gw .1
!
! CABLING (lab topology, exact ports)
!   R1 G0/1   -> S1 F0/5          (F0/5 = 5th port, top row, first 12-port block)
!   S1 F0/4   -> S3 F0/4          (switch to switch, 4th port on each)
!   S1 F0/6   -> PC-A (your PC)   (6th port; becomes the SPAN destination)
!   S3 F0/18  -> PC-C (classmate) (18th port = 6th port of the second block)
!   Console   -> each device in turn. Serials, G0/0, uplink ports: unused.
!   On a 2960 the top row of a block is the ODD ports (1,3,5,...) and the bottom row the EVEN
!   ports (2,4,6,...): F0/5 is top row 3rd socket, F0/6 is directly below it, F0/4 is bottom row 2nd.
!
! STEP 3 OF THE LAB: erase and reload all three first (our Lab 8 configs conflict: VLANs 20/30/90,
!   sub-interfaces, SVIs). Routers: erase startup-config ; reload ; no to save ; no to setup.
!   Switches: erase startup-config ; delete flash:vlan.dat ; reload ; no ; no.
!
! Passwords are the lab's own values (class / cisco), as the brief specifies. "!" lines are comments.
! ============================================================================


! ============================================================================
! 1. R1  (1941 labelled ISP)  - lab Part 1 Step 4
! ============================================================================
enable
configure terminal
no ip domain-lookup
hostname R1
interface GigabitEthernet0/1
 description to S1 F0/5
 ip address 192.168.1.1 255.255.255.0
 no shutdown
 exit
enable secret class
line console 0
 password cisco
 login
 logging synchronous
 exit
line vty 0 4
 password cisco
 login
 transport input telnet
 exit
end
copy running-config startup-config
!
! CHECK: show ip interface brief -> GigabitEthernet0/1 192.168.1.1 up up (once S1 is powered and cabled)


! ============================================================================
! 2. S1  (2960 Plus, marker MG-S1)  - lab Part 1 Step 5
! ============================================================================
enable
configure terminal
no ip domain-lookup
hostname S1
enable secret class
interface Vlan1
 ip address 192.168.1.2 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.1.1
line console 0
 password cisco
 login
 logging synchronous
 exit
line vty 0 15
 password cisco
 login
 exit
end
copy running-config startup-config
!
! CHECK: show ip interface brief -> Vlan1 192.168.1.2 up up ; F0/4 F0/5 F0/6 up
!        ping 192.168.1.1 -> !!!!!


! ============================================================================
! 3. S3  (2960 labelled BNES1)  - lab Part 1 Step 5
! ============================================================================
enable
configure terminal
no ip domain-lookup
hostname S3
enable secret class
interface Vlan1
 ip address 192.168.1.3 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.1.1
line console 0
 password cisco
 login
 logging synchronous
 exit
line vty 0 15
 password cisco
 login
 exit
end
copy running-config startup-config
!
! CHECK: ping 192.168.1.1 ; ping 192.168.1.2 -> !!!!!


! ============================================================================
! 4. PCs  - lab Part 1 Step 2 and Step 6
! ============================================================================
! PC-A (your PC, lab NIC): 192.168.1.254 / 255.255.255.0 / gateway 192.168.1.1
! PC-C (classmate):        192.168.1.10  / 255.255.255.0 / gateway 192.168.1.1
! Windows answers ping only after (PowerShell as Administrator on each PC):
!   Enable-NetFirewallRule -DisplayName "File and Printer Sharing (Echo Request - ICMPv4-In)"
! Step 6 evidence: from PC-A  ping 192.168.1.1 ; ping 192.168.1.2 ; ping 192.168.1.3 ; ping 192.168.1.10
!                  from PC-C  ping 192.168.1.1 ; ping 192.168.1.2 ; ping 192.168.1.3 ; ping 192.168.1.254
! (If PC-C is absent, pings to .10 fail; everything else still passes. A stand-in PC-C: your PC's
!  second NIC into S3 F0/18 with 192.168.1.10, or a bridged VM on that NIC.)
! Do the Step 6 pings BEFORE Part 2: once F0/6 is a SPAN destination it stops passing normal
! traffic, so PC-A can only capture, not ping.


! ============================================================================
! 5. S1 - lab Part 2 Step 1: the SPAN session
! ============================================================================
enable
configure terminal
monitor session 1 source interface FastEthernet0/5
monitor session 1 destination interface FastEthernet0/6
end
show monitor
copy running-config startup-config
!
! CHECK: show monitor ->  Session 1, Type Local Session, Source Ports Both: Fa0/5, Destination Ports: Fa0/6
!        (screenshot this: it is the configuration evidence)
