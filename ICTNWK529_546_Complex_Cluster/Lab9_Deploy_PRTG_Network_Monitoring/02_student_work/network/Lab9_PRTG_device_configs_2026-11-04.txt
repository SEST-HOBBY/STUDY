! Lab 9 PRTG - device configs. Addressing:
!   SB side (classmate): SB router 192.168.0.1 (NTP master), DC1 192.168.0.2, PRTG core 192.168.0.3, SB-S1 .254
!   MG side (you):       MG router 172.16.0.1, DC2 172.16.0.2 (= remote probe), MG-S1 .254
!   WAN: SB S0/0/0 209.165.200.1 - ISP S0/0/0 .2 ; ISP S0/0/1 .17 - MG S0/0/1 .18 ; tunnel 10.10.10.1 / .2
! Replace: class cisco cisco ciscolab Student cisco123
! Crypto needs securityk9 on SB and MG (show license | include securityk9).
! Order: ISP, SB, MG, SB-S1, MG-S1.


! ===== ISP =====
enable
configure terminal
hostname ISP
no ip domain-lookup
enable secret class
line console 0
 password cisco
 login
 logging synchronous
 exit
line vty 0 15
 password cisco
 login
 exit
interface Serial0/0/0
 ip address 209.165.200.2 255.255.255.240
 clock rate 64000
 no shutdown
 exit
interface Serial0/0/1
 ip address 209.165.200.17 255.255.255.240
 clock rate 64000
 no shutdown
 exit
end
write memory


! ===== SB (classmate's router: NTP master, SNMP + syslog to PRTG 192.168.0.3) =====
enable
clock set 10:00:00 4 Nov 2026
configure terminal
hostname SB
no ip domain-lookup
service timestamps log datetime msec
enable secret class
line console 0
 password cisco
 login
 logging synchronous
 exit
line vty 0 15
 password cisco
 login
 exit
interface GigabitEthernet0/0
 ip address 192.168.0.1 255.255.255.0
 no shutdown
 exit
interface Serial0/0/0
 ip address 209.165.200.1 255.255.255.240
 no shutdown
 exit
ip route 0.0.0.0 0.0.0.0 209.165.200.2
crypto isakmp policy 10
 encryption aes 256
 hash sha
 authentication pre-share
 group 5
 exit
crypto isakmp key cisco123 address 209.165.200.18
crypto ipsec transform-set VPN-SET esp-aes 256 esp-sha-hmac
 mode transport
 exit
crypto ipsec profile VPN-PROFILE
 set transform-set VPN-SET
 exit
interface Tunnel0
 ip address 10.10.10.1 255.255.255.252
 ip mtu 1400
 ip tcp adjust-mss 1360
 tunnel source Serial0/0/0
 tunnel destination 209.165.200.18
 tunnel protection ipsec profile VPN-PROFILE
 no shutdown
 exit
router ospf 1
 router-id 1.1.1.1
 passive-interface GigabitEthernet0/0
 network 192.168.0.0 0.0.0.255 area 0
 network 10.10.10.0 0.0.0.3 area 0
 exit
ntp master 5
ntp update-calendar
ip access-list standard SNMP_ACL
 permit host 192.168.0.3
 exit
snmp-server community ciscolab RO SNMP_ACL
snmp-server location SB
snmp-server contact Student
snmp-server host 192.168.0.3 version 2c ciscolab
snmp-server enable traps
logging host 192.168.0.3
logging trap informational
end
write memory
! check: show crypto isakmp sa (QM_IDLE) ; show ip ospf neighbor (FULL on Tunnel0) ; ping 172.16.0.2
