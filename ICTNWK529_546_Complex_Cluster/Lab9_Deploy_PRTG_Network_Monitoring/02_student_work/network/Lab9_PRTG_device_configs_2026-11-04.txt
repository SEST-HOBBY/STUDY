! 2. SB  (Cisco 2911 or 1941 labelled BNE/R1) - NTP master, SNMP agent -> SB-FS2, syslog -> SB-FS2
! ============================================================================
enable

configure terminal
hostname SB
no ip domain-lookup
service password-encryption
service timestamps log datetime msec
clock timezone AEST 10
enable secret <enable-secret>
banner motd # Lab 9 SB - authorised use only #
line console 0
 password <console-password>
 login
 logging synchronous
 exit
line vty 0 15
 password <vty-password>
 login
 exit
!
interface GigabitEthernet0/0
 description Trunk-to-SB-S1-Gi0/2
 no ip address
 no shutdown
 exit
interface GigabitEthernet0/0.20
 description VLAN20-CS
 encapsulation dot1Q 20
 ip address 192.168.1.1 255.255.255.192
 exit
interface GigabitEthernet0/0.30
 description VLAN30-Marketing
 encapsulation dot1Q 30
 ip address 192.168.1.65 255.255.255.192
 exit
interface GigabitEthernet0/0.90
 description VLAN90-Server
 encapsulation dot1Q 90
 ip address 192.168.2.1 255.255.255.240
 exit
interface Serial0/0/0
 description WAN-to-ISP-S0/0/0
 ip address 209.165.200.1 255.255.255.240
 no shutdown
 exit
ip route 0.0.0.0 0.0.0.0 209.165.200.2
!
! --- Lab 5 tunnel: IPsec over GRE (needs securityk9) ---
crypto isakmp policy 10
 encryption aes 256
 hash sha
 authentication pre-share
 group 5
 lifetime 3600
 exit
crypto isakmp key <vpn-psk> address 209.165.200.18
crypto ipsec transform-set VPN-SET esp-aes 256 esp-sha-hmac
 mode transport
 exit
crypto ipsec profile VPN-PROFILE
 set transform-set VPN-SET
 exit
interface Tunnel0
 description GRE-over-IPsec-to-MG
 ip address 10.10.10.1 255.255.255.252
 ip mtu 1400
 ip tcp adjust-mss 1360
 tunnel source Serial0/0/0
 tunnel destination 209.165.200.18
 tunnel protection ipsec profile VPN-PROFILE
 no shutdown
 exit
router ospf 1
 router-id 1.1.1.1
 passive-interface GigabitEthernet0/0.20
 passive-interface GigabitEthernet0/0.30
 passive-interface GigabitEthernet0/0.90
 network 192.168.1.0 0.0.0.63 area 0
 network 192.168.1.64 0.0.0.63 area 0
 network 192.168.2.0 0.0.0.15 area 0
 network 10.10.10.0 0.0.0.3 area 0
 exit
!
! --- Lab 9.2: NTP master (CN-8.1.2.6) ---
ntp master 5
ntp update-calendar
!
! --- Lab 9.2: SNMP agent reporting to the core probe on SB-FS2 (CN-5.2.2.6) ---
ip access-list standard SNMP_ACL
 permit host 192.168.2.5
 exit
snmp-server community <snmp-community> RO SNMP_ACL
snmp-server location SB-site
snmp-server contact <student names>
snmp-server host 192.168.2.5 version 2c <snmp-community>
snmp-server enable traps
!
! --- Lab 9.2: syslog to the PRTG server on SB-FS2 ---
logging host 192.168.2.5
logging trap informational
logging source-interface GigabitEthernet0/0.90
end
write memory
!
! CHECK: show ip interface brief -> Gi0/0 up (unassigned), .20 .30 .90 up, S0/0/0 up/up, Tunnel0 up/up
