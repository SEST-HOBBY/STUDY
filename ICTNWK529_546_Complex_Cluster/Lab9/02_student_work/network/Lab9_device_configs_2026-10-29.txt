! Addresses

!   ISP        S0/0/0 209.165.200.2/28   S0/0/1 209.165.200.17/28

! ============================================================================
! 1. MG router  (Cisco 1941, chassis tape CNS / R2)
! ============================================================================
enable
configure terminal
hostname MG
no ip domain-lookup
service password-encryption
service timestamps log datetime msec
clock timezone AEST 10
enable secret <enable-secret>
banner motd # Lab 9 MG - authorised use only #
line console 0
 password <console-password>
 login
 logging synchronous
 exec-timeout 30 0
 exit
line vty 0 15
 password <vty-password>
 login
 exit
!
interface GigabitEthernet0/1
 description MG-LAN-to-MG-S1-Gi0/1
 ip address 172.16.0.1 255.255.255.0
 no shutdown
 exit
interface Serial0/0/1
 description WAN-to-ISP-S0/0/1
 encapsulation hdlc
 ip address 209.165.200.18 255.255.255.240
 no shutdown
 exit
!
router ospf 1
 router-id 3.3.3.3
 passive-interface default
 no passive-interface Serial0/0/1
 network 172.16.0.0 0.0.0.255 area 0
 network 209.165.200.16 0.0.0.15 area 0
 exit
!
snmp-server community <snmp-ro-community> RO
snmp-server location MG-Site
snmp-server contact <student names>
ntp server 209.165.200.1
logging host 172.16.0.2
logging trap informational
logging source-interface GigabitEthernet0/1
end
write memory
!
! CHECK: show ip interface brief        -> Gi0/1 172.16.0.1 up/up ; S0/0/1 209.165.200.18 up/up
!        show ip ospf neighbor          -> 2.2.2.2 FULL
!        show ip route                  -> O 192.168.2.0/28, 192.168.1.0/26, 192.168.1.64/26, 209.165.200.0/28
!        ping 192.168.2.1 ; ping 172.16.0.2
!        show ntp associations          -> 209.165.200.1 with * (synced) once SB has ntp master; takes 5-10 min
!        show logging                   -> "Logging to 172.16.0.2" listed
! (ChatGPT's "encapsulation ppp" is NOT used: the ISP side is HDLC and a mismatch
!  takes the serial link down. If the brief wants PPP, change BOTH ends together.)

! ============================================================================
! 6. OPTIONAL: IPsec VTI tunnel SB <-> MG  (ONLY if the Lab 9 brief requires a VPN;
!    needs the securityk9 licence on both routers: "show license" must list
!    securityk9 as active, otherwise "crypto" commands are rejected)
! ============================================================================
! --- MG side ---
! configure terminal
! ip route 209.165.200.1 255.255.255.255 209.165.200.17
! crypto isakmp policy 10
!  encryption aes 256
!  hash sha256
!  authentication pre-share
!  group 14
!  lifetime 3600
!  exit
! crypto isakmp key <vpn-preshared-key> address 209.165.200.1
! crypto ipsec transform-set TS50 esp-aes 256 esp-sha256-hmac
!  mode tunnel
!  exit
! crypto ipsec profile CProfile
!  set transform-set TS50
!  exit
! interface Tunnel0
!  description IPSEC-VTI-to-SB
!  ip address 172.16.12.2 255.255.255.252
!  ip ospf cost 10
!  tunnel source Serial0/0/1
!  tunnel destination 209.165.200.1
!  tunnel mode ipsec ipv4
!  tunnel protection ipsec profile CProfile
!  no shutdown
!  exit
! router ospf 1
!  no passive-interface Tunnel0
!  network 172.16.12.0 0.0.0.3 area 0
!  exit
! end
! write memory
! --- SB side (mirror; without it the tunnel never comes up) ---
! configure terminal
! ip route 209.165.200.18 255.255.255.255 209.165.200.2
! crypto isakmp policy 10
!  encryption aes 256
!  hash sha256
!  authentication pre-share
!  group 14
!  lifetime 3600
!  exit
! crypto isakmp key <vpn-preshared-key> address 209.165.200.18
! crypto ipsec transform-set TS50 esp-aes 256 esp-sha256-hmac
!  mode tunnel
!  exit
! crypto ipsec profile CProfile
!  set transform-set TS50
!  exit
! interface Tunnel0
!  description IPSEC-VTI-to-MG
!  ip address 172.16.12.1 255.255.255.252
!  ip ospf cost 10
!  tunnel source Serial0/0/0
!  tunnel destination 209.165.200.18
!  tunnel mode ipsec ipv4
!  tunnel protection ipsec profile CProfile
!  no shutdown
!  exit
! router ospf 1
!  no passive-interface Tunnel0
!  network 172.16.12.0 0.0.0.3 area 0
!  exit
! end
! write memory
! CHECK (both): show crypto isakmp sa -> QM_IDLE ; show crypto ipsec sa -> encaps/decaps rising ;
!               show interfaces tunnel 0 -> up/up ; show ip route -> site routes now via Tunnel0 (cost 10)
! The /32 static routes keep each tunnel endpoint reachable via the serial path so the
! tunnel cannot try to route itself through itself (recursive routing).
